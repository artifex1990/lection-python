**`src/hangman/game.py`** (первая часть)
```python
"""Логика игры «Виселица»."""

import random

from hangman.words import WORDS

MAX_MISTAKES: int = 6
ALPHABET: str = "абвгдеёжзийклмнопрстуфхцчшщъыьэюя"


def choose_word(words: tuple[str, ...] = WORDS) -> str:
    """Возвращает случайное слово из словаря в нижнем регистре."""
    return random.choice(words).lower()


def make_mask(word: str, guessed: set[str]) -> str:
    """Собирает маску слова: открытые буквы + подчёркивания.

    >>> make_mask("питон", {"п", "н"})
    'п _ _ _ н'
    """
    return " ".join(letter if letter in guessed else "_" for letter in word)


def is_won(word: str, guessed: set[str]) -> bool:
    """True, если все буквы слова уже названы."""
    return set(word) <= guessed
```

**Разбираем 3 новые конструкции:**
1. **Генераторное выражение** `letter if letter in guessed else "_" for letter in word` - «для каждой буквы: если угадана - сама буква, иначе прочерк». Читается почти как английский.
2. `" ".join(...)` - склеить всё через пробел.
3. `set(word) <= guessed` - «множество букв слова является подмножеством угаданных». Одна строка вместо цикла - вот сила правильной структуры данных.

**Проверяем прямо сейчас:**

```python
"""ASCII-графика виселицы."""

STAGES: tuple[str, ...] = (
    r"""
     +---+
     |   |
         |
         |
         |
    =====""",
    r"""
     +---+
     |   |
     O   |
         |
         |
    =====""",
    r"""
     +---+
     |   |
     O   |
     |   |
         |
    =====""",
    r"""
     +---+
     |   |
     O   |
    /|   |
         |
    =====""",
    r"""
     +---+
     |   |
     O   |
    /|\  |
         |
    =====""",
    r"""
     +---+
     |   |
     O   |
    /|\  |
    /    |
    =====""",
    r"""
     +---+
     |   |
     O   |
    /|\  |
    / \  |
    =====""",
)


def draw_gallows(mistakes: int) -> str:
    """Возвращает картинку виселицы для заданного числа ошибок."""
    index = min(mistakes, len(STAGES) - 1)
    return STAGES[index]
```

> `r"""..."""` - «сырая» строка: `\` внутри не считается спецсимволом. Иначе `\|` сломает картинку.
> `min(mistakes, len(STAGES) - 1)` - **защита от выхода за границы**. Приучаемся думать «а что если придёт число больше?».

### 6.5. Итерация 3 (20 мин) - валидация и игровой цикл

Добавляем в **`game.py`**:

```python
def validate_letter(raw: str, guessed: set[str]) -> str | None:
    """Проверяет ввод пользователя.

    Возвращает None, если всё в порядке, иначе - текст ошибки.
    """
    letter = raw.strip().lower()

    match letter:
        case "":
            return "Пустой ввод. Введи букву."
        case _ if len(letter) != 1:
            return "Нужна ровно ОДНА буква."
        case _ if letter not in ALPHABET:
            return "Только русские буквы."
        case _ if letter in guessed:
            return f"Букву «{letter}» ты уже называл(а)."
        case _:
            return None
```

> Тип `str | None` (Python 3.10+) читается «строка ИЛИ ничего». Раньше писали `Optional[str]` - сейчас так не пишут.
> `case _ if условие:` - guard-условие в match. Читается сверху вниз, как список правил.

```python
def play_round(word: str) -> bool:
    """Проводит один раунд. Возвращает True при победе."""
    guessed: set[str] = set()
    mistakes: int = 0

    while mistakes < MAX_MISTAKES and not is_won(word, guessed):
        print(draw_gallows(mistakes))
        print(f"\nСлово: {make_mask(word, guessed)}")
        print(f"Ошибки: {mistakes}/{MAX_MISTAKES}")
        if guessed:
            print(f"Названы: {', '.join(sorted(guessed))}")

        raw = input("\nТвоя буква: ")
        error = validate_letter(raw, guessed)

        if error is not None:
            print(f"⚠️  {error}")
            continue

        letter = raw.strip().lower()
        guessed.add(letter)

        if letter in word:
            print(f"✅ Есть такая буква!")
        else:
            mistakes += 1
            print(f"❌ Мимо.")

    print(draw_gallows(mistakes))
    won = is_won(word, guessed)

    if won:
        print(f"\n🎉 Победа! Слово: {word.upper()}")
    else:
        print(f"\n💀 Проигрыш. Слово было: {word.upper()}")

    return won
```

**Обязательно проговорить:**
- `continue` - «пропустить остаток итерации, вернуться к началу цикла». Ошибка ввода **не должна** стоить жизни.
- `while условие1 and not условие2` - цикл идёт, пока есть жизни И слово не отгадано.
- `if error is not None` - для `None` сравниваем через `is`, а не `==`.
- Заметить: `mistakes += 1` находится **внутри** `else`. Забыть отступ = классический баг.

```python
    """Спрашивает да/нет, пока не получит понятный ответ."""
    while True:
        answer = input(f"{question} (да/нет): ").strip().lower()
        match answer:
            case "да" | "д" | "y" | "yes":
                return True
            case "нет" | "н" | "n" | "no":
                return False
            case _:
                print("Не понял. Ответь «да» или «нет».")


def main() -> None:
    """Запускает игровую сессию с подсчётом статистики."""
    print("=" * 40)
    print("🎯  ВИСЕЛИЦА".center(40))
    print("=" * 40)
    print("Угадай слово по буквам. У тебя 6 ошибок.\n")

    stats: dict[str, int] = {"wins": 0, "losses": 0}

    while True:
        word = choose_word()
        if play_round(word):
            stats["wins"] += 1
        else:
            stats["losses"] += 1

        print(f"\n📊 Счёт - победы: {stats['wins']}, поражения: {stats['losses']}")

        if not ask_yes_no("\nСыграем ещё?"):
            break

    print("\nСпасибо за игру! 👋")


if __name__ == "__main__":
    main()
```

**Важный концепт `if __name__ == "__main__":`**
> Это «выключатель». Если файл запустили напрямую - код выполнится. Если его **импортировали** (например, тесты) - не выполнится. Без этой строки тесты будут запускать твою игру. Это одна из первых вещей, которая отличает новичка от того, кто понимает.

### Приложение: альтернативный проект - «Крестики-нолики»

Если группа явно быстрая или хочется вариативности, тот же каркас урока работает на крестиках-ноликах. Ключевые функции:

```python
Board = list[list[str]]  # алиас типа - тоже современная практика


def create_board(size: int = 3) -> Board:
    """Создаёт пустое поле size×size."""
    return [[" "] * size for _ in range(size)]


def render(board: Board) -> str:
    """Рисует поле для консоли."""
    rows = [" | ".join(row) for row in board]
    sep = "\n" + "-" * (len(board) * 4 - 3) + "\n"
    return sep.join(rows)


def winner(board: Board) -> str | None:
    """Возвращает 'X', 'O' или None."""
    lines = [
        *board,  # строки
        *zip(*board),  # столбцы
        [board[i][i] for i in range(len(board))],  # главная диагональ
        [board[i][~i] for i in range(len(board))],  # побочная диагональ
    ]
    for line in lines:
        if len(set(line)) == 1 and line[0] != " ":
            return line[0]
    return None
```

**Плюсы этого варианта:** вложенные списки, `zip(*board)` (распаковка - эффектный момент), два игрока, легко доращивается до бота с алгоритмом минимакс на следующем занятии.
**Минусы:** сложнее для абсолютных новичков (двумерные структуры), поэтому по умолчанию - «Виселица».
